import sys
sys.path.insert(0, "src")
import numpy as np, pandas as pd
from scipy.stats import spearmanr
from fjordvik_service.ml import pipeline, splits, model, anomaly
ev, allow = splits.load_evaluation(), splits.load_allowlist()
cut = splits.period(ev, "validation")["cut_off"]
data = pipeline.load(splits.TABLES_DIR, ev, allow, cut_off=cut); data["allow"] = allow
pipeline.add_iforest(data, ev)
rows, m = data["rows"], data["masks"]
train, val = rows[m["train"]], rows[m["validation_scored"]]
cols = pipeline.feature_columns(allow)
settings = {"max_leaf_nodes": 15, "max_iter": 100, "learning_rate": 0.05}
dummy = val["label"].mean()
# 1. random scores: the noise floor of AP for this many rows
rnd = [model.ap(val["label"], anomaly.rng(100 + k).uniform(size=len(val))) / dummy for k in range(20)]
print("random scores ratio: mean %.3f, min %.3f, max %.3f" % (np.mean(rnd), min(rnd), max(rnd)))
# 2. shuffled model: does its score track how unusual a row is?
out = []
for k in range(1, 6):
    y = anomaly.rng(ev["seed"]["value"] + k).permutation(train["label"].to_numpy())
    f = model.fit(train, cols, settings, ev["seed"]["value"], labels=y)
    sc = model.score(f, val, cols)
    rho = spearmanr(sc, val["iforest_score"]).statistic
    out.append((round(model.ap(val["label"], sc) / dummy, 3), int(f.n_iter_), round(float(np.std(sc)), 5), round(rho, 3)))
print("shuffled: (ratio, iterations, score sd, rank corr with forest score)", out)
# 3. same shuffles, pump facts only (no telemetry)
facts = allow["pump_facts_only_model"]["columns"]
fo = []
for k in range(1, 6):
    y = anomaly.rng(ev["seed"]["value"] + k).permutation(train["label"].to_numpy())
    f = model.fit(train, facts, settings, ev["seed"]["value"], labels=y)
    fo.append(round(model.ap(val["label"], model.score(f, val, facts)) / dummy, 3))
print("shuffled, pump facts only:", fo)

import numpy as np, pandas as pd

ev, allow = splits.load_evaluation(), splits.load_allowlist()
cut = splits.period(ev, "validation")["cut_off"]
data = pipeline.load(splits.TABLES_DIR, ev, allow, cut_off=cut); data["allow"] = allow
pipeline.add_iforest(data, ev)
rows, m = data["rows"], data["masks"]
train, val = rows[m["train"]].copy(), rows[m["validation_scored"]].copy()
cols = pipeline.feature_columns(allow)
settings = {"max_leaf_nodes": 15, "max_iter": 100, "learning_rate": 0.05}
y = anomaly.rng(ev["seed"]["value"] + 5).permutation(train["label"].to_numpy())
f = model.fit(train, cols, settings, ev["seed"]["value"], labels=y)
val["s"] = model.score(f, val, cols)
print("distinct scores", val["s"].nunique(), "of", len(val))
top = val.sort_values("s", ascending=False)
for n in [50, 200, 1000, 5000]:
    t = top.head(n)
    print(n, "positive share %.3f" % t["label"].mean(), "pumps", t["asset_id"].nunique(), "median z_max %.2f" % t["z_max"].median(),
          "median age %.1f" % t["age_years"].median())
print("all: positive share %.3f, median z_max %.2f, median age %.1f" % (val["label"].mean(), val["z_max"].median(), val["age_years"].median()))
print(top.head(200).groupby("asset_id").agg(rows=("s", "size"), pos=("label", "sum")).sort_values("rows", ascending=False).head(8))
# train: shuffled positives in the region the model ranks highest
